# 机器管理模块实现计划

## 模块概述

机器管理模块包含 2 个功能点,用于管理和监控机器实例。

## 功能列表

| 功能 ID | 功能名称 | 优先级 | 状态 | 说明 |
|---------|----------|--------|------|------|
| P1-M14 | 机器健康检查 | P1 | 需增强 | 需要添加健康检查 API |
| P1-M16 | 编辑机器配置 | P1 | 需实现 | 需要添加编辑功能 |

## 现有实现分析

### 已实现的功能

1. **机器列表页面**: `src/views/pages/machines.ejs`
2. **机器 Model**: `src/models/machine.model.ts`
3. **基础 CRUD**: 创建、查询、更新、删除
4. **批量操作**: 批量重启

### 已有的 API

1. **GET /admin/machines** - 机器列表页面
2. **POST /api/admin/machines** - 添加机器
3. **DELETE /api/machines/:id** - 删除机器
4. **POST /api/machines/:id/restart** - 重启机器

### 需要增强的功能

1. **机器健康检查**: 检查机器连接状态、资源使用情况
2. **编辑机器配置**: 修改机器的主机名、IP、端口等配置

```typescript
// 获取机器详情
static async getDetailById(id: string): Promise<(MachineInfo & {
  activeSessions: number;
  healthStatus?: string;
}) | null> {
  try {
    const machine = await this.findById(id);
    if (!machine) return null;

    // 获取活跃会话数
    const { SessionModel } = await import('./session.model.js');
    const activeSessions = await SessionModel.paginate(1, 999, {
      filters: {
        status: 'active'
      }
    });

    // 过滤出该机器的活跃会话
    const machineActiveSessions = activeSessions.items.filter(
      (s: any) => s.machine_id === id
    ).length;

    return {
      ...machine,
      activeSessions: machineActiveSessions
    };
  } catch (error) {
    console.error('获取机器详情失败:', error);
    return null;
  }
}

// 执行健康检查
static async healthCheck(id: string): Promise<{
  machineId: string;
  status: 'healthy' | 'unhealthy';
  grpcConnected: boolean;
  responseTime?: number;
  activeInstances?: number;
  systemInfo?: {
    cpuUsage: number;
    memoryUsage: number;
    diskUsage: number;
    uptime?: number;
  };
  error?: string;
  checkedAt: Date;
}> {
  try {
    const machine = await this.findById(id);
    if (!machine) {
      return {
        machineId: id,
        status: 'unhealthy',
        grpcConnected: false,
        error: '机器不存在',
        checkedAt: new Date()
      };
    }

    // 导入 gRPC 客户端
    const { machineGrpcService } = await import('../services/machine-grpc.service.js');

    const startTime = Date.now();
    try {
      // 尝试连接并获取状态
      const health = await machineGrpcService.checkMachineHealth(id);
      const responseTime = Date.now() - startTime;

      return {
        machineId: id,
        status: 'healthy',
        grpcConnected: true,
        responseTime,
        activeInstances: health.activeInstances,
        systemInfo: {
          cpuUsage: health.cpuUsage,
          memoryUsage: health.memoryUsage,
          diskUsage: health.diskUsage,
          uptime: health.uptime
        },
        checkedAt: new Date()
      };
    } catch (grpcError: any) {
      return {
        machineId: id,
        status: 'unhealthy',
        grpcConnected: false,
        error: grpcError.message || 'gRPC 连接失败',
        checkedAt: new Date()
      };
    }
  } catch (error: any) {
    console.error('健康检查失败:', error);
    return {
      machineId: id,
      status: 'unhealthy',
      grpcConnected: false,
      error: error.message || '健康检查失败',
      checkedAt: new Date()
    };
  }
}

// 批量健康检查
static async batchHealthCheck(ids: string[]): Promise<Array<{
  machineId: string;
  status: 'healthy' | 'unhealthy';
  error?: string;
}>> {
  const results = await Promise.all(
    ids.map(async (id) => {
      const result = await this.healthCheck(id);
      return {
        machineId: id,
        status: result.status,
        error: result.error
      };
    })
  );

  return results;
}
```

### 步骤 2: 创建机器控制器

**文件**: `src/controllers/machine.controller.ts`

```typescript
import { FastifyReply } from 'fastify';
import { MachineModel, UpdateMachineInput } from '../models/machine.model.js';
import { sendSuccess, sendError } from '../utils/response.js';
import { AuthenticatedRequestWithParams } from '@shared/types/index.js';

// 获取机器详情
export async function getMachineById(request: AuthenticatedRequestWithParams<{ id: string }>, reply: FastifyReply) {
  try {
    const machineId = request.params.id;

    const machine = await MachineModel.getDetailById(machineId);
    if (!machine) {
      return sendError(reply, '机器不存在', 404);
    }

    return sendSuccess(reply, machine);
  } catch (error) {
    request.log.error(error);
    return sendError(reply, '获取机器详情失败', 500);
  }
}

// 更新机器配置
export async function updateMachine(request: AuthenticatedRequestWithParams<{ id: string }>, reply: FastifyReply) {
  try {
    const machineId = request.params.id;
    const updateData = request.body as UpdateMachineInput;

    // 检查机器是否存在
    const existingMachine = await MachineModel.findById(machineId);
    if (!existingMachine) {
      return sendError(reply, '机器不存在', 404);
    }

    // 更新机器
    const updatedMachine = await MachineModel.update(machineId, updateData);
    if (!updatedMachine) {
      return sendError(reply, '更新机器失败', 500);
    }

    return sendSuccess(reply, updatedMachine);
  } catch (error) {
    request.log.error(error);
    return sendError(reply, '更新机器失败', 500);
  }
}

// 健康检查
export async function healthCheck(request: AuthenticatedRequestWithParams<{ id: string }>, reply: FastifyReply) {
  try {
    const machineId = request.params.id;

    const result = await MachineModel.healthCheck(machineId);
    return sendSuccess(reply, result);
  } catch (error) {
    request.log.error(error);
    return sendError(reply, '健康检查失败', 500);
  }
}

// 批量健康检查
export async function batchHealthCheck(request: AuthenticatedRequest, reply: FastifyReply) {
  try {
    const { machineIds } = request.body as { machineIds: string[] };

    if (!Array.isArray(machineIds) || machineIds.length === 0) {
      return sendError(reply, '无效的机器 ID 列表', 400);
    }

    const results = await MachineModel.batchHealthCheck(machineIds);

    const healthy = results.filter(r => r.status === 'healthy').length;
    const unhealthy = results.filter(r => r.status === 'unhealthy').length;

    return sendSuccess(reply, {
      total: results.length,
      healthy,
      unhealthy,
      results
    });
  } catch (error) {
    request.log.error(error);
    return sendError(reply, '批量健康检查失败', 500);
  }
}

export default {
  getMachineById,
  updateMachine,
  healthCheck,
  batchHealthCheck
};
```

```html
              <div><span class="text-gray-500">内存:</span> ${data.systemInfo.memoryUsage}%</div>
              <div><span class="text-gray-500">磁盘:</span> ${data.systemInfo.diskUsage}%</div>
              <div><span class="text-gray-500">活跃实例:</span> ${data.activeInstances || '-'}</div>
            </div>
          ` : ''}
          ${data.error ? `<div class="text-red-600 text-sm mt-2">错误: ${data.error}</div>` : ''}
        `;
      }
    } catch (error) {
      alert('健康检查失败: ' + error.message);
    } finally {
      healthCheckBtn.disabled = false;
      healthCheckBtn.innerHTML = '<i class="fas fa-heartbeat mr-1"></i> 检查';
    }
  });
</script>
```

```typescript
// 机器详情页
fastify.get('/admin/machines/:id', {
  preHandler: [authenticate, verifyAdmin]
}, async (request, reply) => {
  try {
    const { MachineModel } = await import('../models/machine.model.js');
    const machine = await MachineModel.getDetailById(
      (request.params as any).id
    );

    if (!machine) {
      return reply.status(404).view('404.ejs');
    }

    return reply.view('pages/machine-detail.ejs', {
      user: request.user,
      machine,
      currentPage: 'machines'
    });
  } catch (error) {
    request.log.error('获取机器详情失败:', error);
    return reply.status(500).send('获取机器详情失败');
  }
});
```
