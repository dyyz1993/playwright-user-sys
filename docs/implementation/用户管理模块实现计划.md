```typescript
// 带搜索和筛选的分页查询
static async paginate(
  page: number = 1,
  limit: number = 20,
  options?: {
    sort?: string;
    order?: 'asc' | 'desc';
    search?: string;
    role?: string;
    status?: string;
  }
): Promise<PaginatedResponse<User>> {
  try {
    const offset = (page - 1) * limit;
    const sort = options?.sort || 'created_at';
    const order = options?.order || 'desc';

    // 构建查询
    let query = db('users').select(
      'id',
      'username',
      'email',
      'role',
      'status',
      'credits',
      'created_at'
    );

    // 搜索 - 模糊匹配用户名或邮箱
    if (options?.search) {
      const searchTerm = `%${options.search}%`;
      query = query.where(function() {
        this.where('username', 'like', searchTerm)
          .orWhere('email', 'like', searchTerm);
      });
    }

    // 角色筛选
    if (options?.role) {
      query = query.where('role', options.role);
    }

    // 状态筛选
    if (options?.status) {
      query = query.where('status', options.status);
    }

    // 执行查询
    const [users, totalResult] = await Promise.all([
      query.clone()
        .orderBy(sort, order)
        .limit(limit)
        .offset(offset),
      query.count('id as count').first()
    ]);

    return {
      items: users,
      total: totalResult ? Number(totalResult.count) : 0,
      page,
      limit,
      totalPages: Math.ceil((totalResult ? Number(totalResult.count) : 0) / limit)
    };
  } catch (error) {
    console.error('分页查询用户失败:', error);
    return {
      items: [],
      total: 0,
      page,
      limit,
      totalPages: 0
    };
  }
}
```

```typescript
// 获取所有用户(增强版)
export async function getAllUsers(request: AuthenticatedRequest, reply: FastifyReply) {
  try {
    const query = request.query as any;
    const page = parseInt(query.page) || 1;
    const limit = parseInt(query.limit) || 20;
    const sort = query.sort || 'created_at';
    const order = query.order || 'desc';
    const search = query.search;
    const role = query.role;
    const status = query.status;

    const users = await UserModel.paginate(page, limit, {
      sort,
      order,
      search,
      role,
      status
    });

    // 移除敏感信息
    const sanitizedUsers = users.items.map(user => ({
      id: user.id,
      username: user.username,
      email: user.email,
      role: user.role,
      status: user.status,
      credits: user.credits,
      created_at: user.created_at,
    }));

    return sendPaginated(reply, {
      ...users,
      items: sanitizedUsers,
    });
  } catch (error) {
    request.log.error(error);
    return sendError(reply, '获取用户列表失败', 500);
  }
}
```

```javascript
const searchInput = document.getElementById('search-users');
let searchTimeout = null;

searchInput?.addEventListener('input', () => {
  // 清除之前的定时器
  if (searchTimeout) {
    clearTimeout(searchTimeout);
  }

  // 延迟 500ms 后执行搜索

roleFilter?.addEventListener('change', () => {
  const selectedRole = roleFilter.value;
  loadUsers({ role: selectedRole, page: 1 });
});

// 加载用户列表
async function loadUsers(params = {}) {
  const queryParams = new URLSearchParams();

  // 设置默认参数
  if (!params.page) params.page = 1;
  if (!params.limit) params.limit = 20;
  if (!params.sort) params.sort = 'created_at';
  if (!params.order) params.order = 'desc';

  Object.keys(params).forEach(key => {
    if (params[key] !== '' && params[key] !== null && params[key] !== undefined) {
      queryParams.set(key, params[key]);
    }
  });

  const response = await fetch(`/api/admin/users?${queryParams}`);
  const result = await response.json();

  if (result.success) {
    renderUsers(result.data.items);
    updatePagination(result.data);
  }
}
```

```typescript
// 搜索用户建议
export async function suggestUsers(request: AuthenticatedRequest, reply: FastifyReply) {
  try {
    const query = request.query as any;
    const searchTerm = query.q;
    const limit = parseInt(query.limit) || 10;

    if (!searchTerm || searchTerm.length < 2) {
      return sendSuccess(reply, []);
    }

    const users = await db('users')
      .select('id', 'username', 'email')
      .where('username', 'like', `%${searchTerm}%`)
      .orWhere('email', 'like', `%${searchTerm}%`)
      .limit(limit);

    return sendSuccess(reply, users);
  } catch (error) {
    request.log.error(error);
    return sendError(reply, '搜索用户失败', 500);
  }
}
```
