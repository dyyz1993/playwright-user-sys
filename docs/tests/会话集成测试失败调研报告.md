### 二、并发竞争条件问题 (2个测试)

#### 5. S-CONCURRENCY-04: 并发操作同一会话应该幂等
- **错误**: `expected 97 to be 99`
- **根本原因**: 并发调用 `markDisconnected` 时，多次扣除用户积分
- **问题代码**:
  ```typescript
  // src/models/session.model.ts:198-299
  static async markDisconnected(id: string, duration: number) {
    // ... 省略代码

    // 问题: 即使已有credits_used，仍然可能重复扣费
    if (newCreditsUsed > previousCreditsUsed) {
      const creditsToDeduct = newCreditsUsed - previousCreditsUsed;
      await UserModel.deductCredits(session.user_id, creditsToDeduct);
    }

    // 并发调用时，多个请求可能同时看到 credits_used=0
    // 导致多次扣费
  }
  ```
- **测试场景**:
  ```typescript
  // 3个并发请求同时释放同一会话
  const results = await Promise.all([
    SessionModel.markDisconnected(session.id, 60),
    SessionModel.markDisconnected(session.id, 60),
    SessionModel.markDisconnected(session.id, 60),
  ]);
  // 期望: 用户从100点变成99点 (只扣1次)
  // 实际: 用户从100点变成97点 (扣了3次)
  ```
- **修复难度**: 中 - 需要实现数据库级别的乐观锁或悲观锁
- **优先级**: P0 - 可能导致财务问题
- **修复建议**:

  **方案1: 使用数据库事务 + 行锁**
  ```typescript
  static async markDisconnected(id: string, duration: number) {
    return db.transaction(async (trx) => {
      // 使用 SELECT FOR UPDATE 锁定行
      const session = await trx('sessions')
        .where({ id })
        .forUpdate()
        .first();

      if (!session) return null;

      // 检查是否已经扣费
      if (session.credits_used > 0) {
        // 已扣费，只更新状态
        return await updateSession(trx, id, { ... });
      }

      // 计算扣费并更新
      const creditsUsed = Math.max(1, Math.ceil(duration / 60));
      await updateSession(trx, id, { credits_used: creditsUsed, ... });
      await UserModel.deductCredits(trx, session.user_id, creditsUsed);
    });
  }
  ```

  **方案2: 使用乐观锁（推荐）**
  ```typescript
  static async markDisconnected(id: string, duration: number) {
    const maxRetries = 3;
    for (let i = 0; i < maxRetries; i++) {
      try {
        const session = await this.findById(id);
        if (!session) return null;

        if (session.credits_used > 0) {
          // 已扣费，只更新状态
          await db('sessions').where({ id }).update({
            status: SessionStatus.DISCONNECTED,
            duration: Math.max(session.duration, duration),
          });
          return this.findById(id);
        }

        // 尝试更新（乐观锁）
        const creditsUsed = Math.max(1, Math.ceil(duration / 60));
        const updated = await db('sessions')
          .where({ id, credits_used: 0 })  // 只有 credits_used=0 时才更新
          .update({
            status: SessionStatus.DISCONNECTED,
            duration,
            credits_used: creditsUsed,
          });

        if (updated > 0) {
          // 更新成功，扣费
          await UserModel.deductCredits(session.user_id, creditsUsed);
          return this.findById(id);
        } else {
          // 更新失败（已被其他事务修改），重试
          continue;
        }
      } catch (error) {
        if (i === maxRetries - 1) throw error;
        await new Promise(resolve => setTimeout(resolve, 10 * (i + 1)));
      }
    }
  }
  ```

```typescript
// src/models/session.model.ts
static async markDisconnected(id: string, duration: number) {
  const maxRetries = 3;

  for (let attempt = 0; attempt < maxRetries; attempt++) {
    try {
      const { logger } = await import('@shared/utils/logger.js');

      // 获取会话
      const session = await this.findById(id);
      if (!session) {
        logger.error(`标记会话已断开失败: 会话不存在 (${id})`);
        return null;
      }

      // 如果已经是终态，直接返回
      if (session.status === SessionStatus.DISCONNECTED ||
          session.status === SessionStatus.EXPIRED ||
          session.status === SessionStatus.ERROR) {
        return session;
      }

      let finalDuration = duration;
      if (finalDuration === 0 && session.start_time) {
        const now = new Date();
        const startTime = new Date(session.start_time);
        finalDuration = Math.floor((now.getTime() - startTime.getTime()) / 1000);
      }

      // 确保duration不为负数
      if (finalDuration < 0) {
        finalDuration = 0;
      }

      const creditsUsed = finalDuration >= 0 ? Math.max(1, Math.ceil(finalDuration / 60)) : 0;

      // 如果已经扣费，只更新状态不扣费
      if (session.credits_used > 0) {
        const newDuration = Math.max(session.duration, duration);
        const newCreditsUsed = Math.max(session.credits_used, creditsUsed);

        await db('sessions').where({ id }).update({
          status: SessionStatus.DISCONNECTED,
          end_time: new Date(),
          duration: newDuration,
          credits_used: newCreditsUsed,
          updated_at: new Date(),
        });

        return this.findById(id);
      }

      // 乐观锁: 只有 credits_used=0 时才更新并扣费
      const affectedRows = await db('sessions')
        .where({ id, credits_used: 0 })
        .update({
          status: SessionStatus.DISCONNECTED,
          end_time: new Date(),
          duration: finalDuration,
          credits_used: creditsUsed,
          updated_at: new Date(),
        });

      if (affectedRows === 0) {
        // 更新失败，可能是其他事务已经更新，重试
        if (attempt < maxRetries - 1) {
          await new Promise(resolve => setTimeout(resolve, 10 * (attempt + 1)));
          continue;
        }
        // 最后一次尝试仍然失败，返回当前状态
        return this.findById(id);
      }

      // 更新成功，扣除用户积分
      const { UserModel } = await import('./user.model.js');
      await UserModel.deductCredits(session.user_id, creditsUsed);

      return this.findById(id);

    } catch (error) {
      if (attempt === maxRetries - 1) {
        throw error;
      }
      await new Promise(resolve => setTimeout(resolve, 10 * (attempt + 1)));
    }
  }

  return null;
}
```
