// 添加状态筛选支持
static async paginate(
  page: number = 1,
  limit: number = 10,
  options: { status?: string } = {}
): Promise<PaginatedResponse<Session & { username?: string }>> {
  try {
    const offset = (page - 1) * limit;

    let query = db('sessions')
      .select('sessions.*', 'users.username')
      .leftJoin('users', 'sessions.user_id', 'users.id');

    // 添加状态筛选
    if (options.status) {
      const statusMap: Record<string, string[]> = {
        'active': [SessionStatus.CREATED, SessionStatus.CONNECTED],
        'ended': [SessionStatus.DISCONNECTED, SessionStatus.EXPIRED],
        'error': [SessionStatus.ERROR],
      };
      const statuses = statusMap[options.status] || [options.status];
      query = query.whereIn('sessions.status', statuses);
    }

    const [sessions, total] = await Promise.all([
      query.orderBy('sessions.created_at', 'desc')
        .limit(limit)
        .offset(offset),
      db('sessions').count('id as count').first(),
    ]);

    return {
      items: sessions.map(/* ... */),
      total: total ? Number(total.count) : 0,
      page,
      limit,
      totalPages: Math.ceil((total ? Number(total.count) : 0) / limit),
    };
  } catch (error) {
    console.error('分页查询会话失败:', error);
    return {
      items: [],
      total: 0,
      page,
      limit,
      totalPages: 0,
    };
  }
}
```

async createMachine(req: FastifyRequest, reply: FastifyReply) {
  try {
    const adminId = req.user?.id;
    const body = req.body as any;

    // 验证输入
    if (!body.hostname || !body.ip) {
      return reply.status(400).send({ success: false, error: '主机名和IP地址不能为空' });
    }

    // 生成机器ID
    const machineId = uuidv4();

    // 创建机器
    const { MachineModel } = await import('../models/machine.model.js');
    const machine = await MachineModel.register({
      id: machineId,
      hostname: body.hostname,
      ip: body.ip,
      grpcPort: body.grpcPort,
      proxyPort: body.proxyPort,
      maxInstances: body.maxInstances || 10,
    });

    if (!machine) {
      return reply.status(500).send({ success: false, error: '创建机器失败' });

    }).catch(err => req.log.error('记录操作日志失败:', err));

    return reply.send({
      success: true,
      message: '机器创建成功',
      data: machine,
    });
  } catch (error: any) {
    req.log.error('创建机器失败:', error);

    const body = req.body as any;

    if (!body.ids || !Array.isArray(body.ids) || body.ids.length === 0) {
      return reply.status(400).send({ success: false, error: '无效的用户ID列表' });
    }

    const { UserModel } = await import('../models/user.model.js');
    const { OperationLogModel } = await import('../models/operation-log.model.js');

    let deletedCount = 0;
    const errors: string[] = [];

    for (const userId of body.ids) {
      try {
        const user = await UserModel.findById(userId);
        if (!user) {
          errors.push(`用户ID ${userId} 不存在`);
          continue;
        }

        if (user.role === UserRole.ADMIN) {
          errors.push(`不允许删除管理员: ${user.username}`);
          continue;
        }

        await UserModel.delete(userId);
        deletedCount++;

        // 记录操作日志
        OperationLogModel.create({
          admin_id: adminId || 0,
          action: '批量删除用户',
          details: { username: user.username },
          target_user_id: userId,
        }).catch(err => req.log.error('记录操作日志失败:', err));
      } catch (error) {
        errors.push(`删除用户ID ${userId} 失败`);
      }
    }

    return reply.send({
      success: true,
      message: `成功删除 ${deletedCount} 个用户`,
      data: {
        deletedCount,
        totalRequested: body.ids.length,
        errors,
      },
    });
  } catch (error: any) {
    req.log.error('批量删除用户失败:', error);
    return reply.status(500).send({ success: false, error: '批量删除用户失败: ' + error.message });
  }
}
```

#### 接口实现

```typescript
// src/controllers/user.controller.ts
async exportUsers(req: FastifyRequest, reply: FastifyReply) {
  try {
    const query = req.query as { format?: string };

    const { UserModel } = await import('../models/user.model.js');
    const users = await UserModel.getAll();

    if (query.format === 'csv') {
      // 生成 CSV
      const headers = ['ID', '用户名', '邮箱', '角色', '状态', '积分', '创建时间'];
      const rows = users.map(u => [
        u.id,
        u.username,
        u.email || '',
        u.role,
        u.status,
        u.credits,
        u.created_at,
      ]);

      let csv = headers.join(',') + '\n';
      csv += rows.map(row => row.map(cell => `"${cell}"`).join(',')).join('\n');

      reply.header('Content-Type', 'text/csv; charset=utf-8');
      reply.header('Content-Disposition', `attachment; filename="users_${Date.now()}.csv"`);
      return reply.send('\uFEFF' + csv); // 添加 BOM 以支持中文
    }

    return reply.status(400).send({ success: false, error: '不支持的导出格式' });
  } catch (error: any) {
    req.log.error('导出用户失败:', error);
    return reply.status(500).send({ success: false, error: '导出用户失败: ' + error.message });
  }
}
```
