## 测试实现

### 完整测试文件结构

```typescript
import { describe, it, expect, beforeAll, afterAll } from 'vitest';
import { FastifyInstance } from 'fastify';
import { build } from '../../helpers/app.js';
import { UserModel } from '../../../models/user.model.js';
import { clearAllTables } from '../../helpers/database.js';
import { createTestUser, createTestAdmin } from '../../helpers/factories.js';
import { generateToken } from '../../../utils/auth.js';
import { z } from 'zod';

describe('用户管理 API 集成测试', () => {
  let app: FastifyInstance;
  let testAdmin: any;
  let testUser: any;
  let adminToken: string;
  let userToken: string;

  beforeAll(async () => {
    await clearAllTables();
    app = await build();
    testAdmin = await createTestAdmin();
    testUser = await createTestUser();
    adminToken = generateToken(testAdmin);
    userToken = generateToken(testUser);
  });

  afterAll(async () => {
    await clearAllTables();
    await app.close();
  });

  // ==================== A. 用户管理核心功能 ====================
  describe('A. 用户管理核心功能', () => {

    describe('POST /api/admin/users - 创建用户', () => {

      it('A-01: 管理员创建用户应该成功', async () => {
        const newUser = {
          username: `test_${Date.now()}`,
          password: 'password123',
          email: `test_${Date.now()}@example.com`,
        };

        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: newUser,
        });

        expect(response.statusCode).toBe(201);
        const result = JSON.parse(response.payload);
        expect(result.success).toBe(true);
        expect(result.data).toHaveProperty('id');
        expect(result.data).toHaveProperty('username', newUser.username);
        expect(result.data).not.toHaveProperty('password'); // 密码不应返回
      });

      it('A-02: 创建用户时缺少必填字段应该返回400', async () => {
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: { username: 'test' }, // 缺少 password
        });

        expect(response.statusCode).toBe(400);
      });

      it('A-03: 创建用户时用户名重复应该返回409', async () => {
        const username = `duplicate_${Date.now()}`;

        // 创建第一个用户
        await UserModel.create({
          username,
          password: await hashPassword('pass123'),
          email: 'test@example.com',
        });

        // 尝试创建重复用户名的用户
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: {
            username,
            password: 'pass123',
            email: 'another@example.com',
          },
        });

        expect(response.statusCode).toBe(409);
      });
    });

    describe('GET /api/admin/users - 获取用户列表', () => {

      it('A-08: 获取用户列表（第一页）应该成功', async () => {
        const response = await app.inject({
          method: 'GET',
          url: '/api/admin/users?page=1&limit=10',
          headers: { Authorization: `Bearer ${adminToken}` },
        });

        expect(response.statusCode).toBe(200);
        const result = JSON.parse(response.payload);
        expect(result.success).toBe(true);
        expect(result.data).toHaveProperty('items');
        expect(result.data).toHaveProperty('total');
        expect(result.data).toHaveProperty('page');
        expect(result.data.items).toBeInstanceOf(Array);
      });

      it('A-10: 获取用户列表时搜索用户名应该成功', async () => {
        const response = await app.inject({
          method: 'GET',
          url: `/api/admin/users?search=${testUser.username}`,
          headers: { Authorization: `Bearer ${adminToken}` },
        });

        expect(response.statusCode).toBe(200);
        const result = JSON.parse(response.payload);
        expect(result.data.items.length).toBeGreaterThan(0);
      });
    });
  });

  // ==================== B. 字段验证测试 ====================
  describe('B. 字段验证测试', () => {

    describe('username 字段验证', () => {

      it('B-username-01: username 少于3字符应该返回400', async () => {
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: {
            username: 'ab', // 少于3字符
            password: 'password123',
          },
        });

        expect(response.statusCode).toBe(400);
      });

      it('B-username-02: username 多于50字符应该返回400', async () => {
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: {
            username: 'a'.repeat(51), // 多于50字符
            password: 'password123',
          },
        });

        expect(response.statusCode).toBe(400);
      });

      it('B-username-03: username 为空应该返回400', async () => {
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: {
            username: '',
            password: 'password123',
          },
        });

        expect(response.statusCode).toBe(400);
      });
    });

    describe('email 字段验证', () => {

      it('B-email-01: 无效邮箱格式应该返回400', async () => {
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: {
            username: `test_${Date.now()}`,
            password: 'password123',
            email: 'invalid-email', // 无效格式
          },
        });

        expect(response.statusCode).toBe(400);
      });

      it('B-email-02: 不传 email 应该成功（使用默认值）', async () => {
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: {
            username: `test_${Date.now()}`,
            password: 'password123',
            // 不传 email
          },
        });

        expect(response.statusCode).toBe(201);
      });
    });

    describe('password 字段验证', () => {

      it('B-password-01: password 少于6字符应该返回400', async () => {
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: {
            username: `test_${Date.now()}`,
            password: '12345', // 少于6字符
          },
        });

        expect(response.statusCode).toBe(400);
      });

      it('B-password-02: 创建后密码不应在响应中返回', async () => {
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: {
            username: `test_${Date.now()}`,
            password: 'password123',
          },
        });

        const result = JSON.parse(response.payload);
        expect(result.data).not.toHaveProperty('password');
      });

      it('B-password-03: 数据库中密码应被哈希存储', async () => {
        const plainPassword = 'password123';
        const user = await UserModel.create({
          username: `test_${Date.now()}`,
          password: await hashPassword(plainPassword),
          email: 'test@example.com',
        });

        const dbUser = await UserModel.findById(user.id);
        expect(dbUser.password).not.toBe(plainPassword); // 应该是哈希值
        expect(dbUser.password).toHaveLength(64); // SHA256 哈希长度
      });
    });

    describe('role 字段验证', () => {

      it('B-role-01: 创建时指定 role=admin 应该成功', async () => {
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: {
            username: `admin_${Date.now()}`,
            password: 'password123',
            role: 'admin',
          },
        });

        expect(response.statusCode).toBe(201);
        const result = JSON.parse(response.payload);
        expect(result.data.role).toBe('admin');
      });

      it('B-role-02: 不传 role 应该使用默认值 user', async () => {
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: {
            username: `user_${Date.now()}`,
            password: 'password123',
            // 不传 role
          },
        });

        expect(response.statusCode).toBe(201);
        const result = JSON.parse(response.payload);
        expect(result.data.role).toBe('user');
      });

      it('B-role-03: 无效 role 值应该返回400', async () => {
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: {
            username: `test_${Date.now()}`,
            password: 'password123',
            role: 'superadmin', // 无效值
          },
        });

        expect(response.statusCode).toBe(400);
      });
    });

    describe('credits 字段验证', () => {

      it('B-credits-01: credits 为负数应该返回400', async () => {
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: {
            username: `test_${Date.now()}`,
            password: 'password123',
            credits: -10, // 负数
          },
        });

        expect(response.statusCode).toBe(400);
      });

      it('B-credits-02: 不传 credits 应该使用默认值 0', async () => {
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: {
            username: `test_${Date.now()}`,
            password: 'password123',
            // 不传 credits
          },
        });

        expect(response.statusCode).toBe(201);
        const result = JSON.parse(response.payload);
        expect(result.data.credits).toBe(0);
      });
    });
  });

  // ==================== C. 业务逻辑测试 ====================
  describe('C. 业务逻辑测试', () => {

    describe('权限控制', () => {

      it('C-auth-01: 非管理员创建用户应该返回403', async () => {
        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users',
          headers: { Authorization: `Bearer ${userToken}` }, // 普通用户
          payload: {
            username: `test_${Date.now()}`,
            password: 'password123',
          },
        });

        expect(response.statusCode).toBe(403);
      });

      it('C-auth-02: 未认证访问应该返回401', async () => {
        const response = await app.inject({
          method: 'GET',
          url: '/api/admin/users',
          // 不传 Authorization
        });

        expect(response.statusCode).toBe(401);
      });

      it('C-auth-03: 无效 token 应该返回401', async () => {
        const response = await app.inject({
          method: 'GET',
          url: '/api/admin/users',
          headers: { Authorization: 'Bearer invalid_token' },
        });

        expect(response.statusCode).toBe(401);
      });
    });

    describe('数据操作', () => {

      it('C-data-01: 删除用户应该从数据库移除', async () => {
        const user = await createTestUser();

        await app.inject({
          method: 'DELETE',
          url: `/api/admin/users/${user.id}`,
          headers: { Authorization: `Bearer ${adminToken}` },
        });

        const deleted = await UserModel.findById(user.id);
        expect(deleted).toBeUndefined();
      });

      it('C-data-02: 更新用户应该修改数据库记录', async () => {
        const user = await createTestUser();
        const newEmail = `updated_${Date.now()}@example.com`;

        await app.inject({
          method: 'PUT',
          url: `/api/admin/users/${user.id}`,
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: { email: newEmail },
        });

        const updated = await UserModel.findById(user.id);
        expect(updated.email).toBe(newEmail);
      });

      it('C-data-03: 添加积分应该增加用户积分', async () => {
        const user = await createTestUser({ credits: 100 });

        await app.inject({
          method: 'POST',
          url: `/api/admin/users/${user.id}/credits`,
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: { amount: 50 },
        });

        const updated = await UserModel.findById(user.id);
        expect(updated.credits).toBe(150);
      });
    });

    describe('批量操作', () => {

      it('C-batch-01: 批量删除应该删除多个用户', async () => {
        const users = await createTestUsers(3);
        const userIds = users.map(u => u.id);

        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users/batch-delete',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: { userIds },
        });

        expect(response.statusCode).toBe(200);

        for (const id of userIds) {
          const user = await UserModel.findById(id);
          expect(user).toBeUndefined();
        }
      });

      it('C-batch-02: 批量充值应该给多个用户添加积分', async () => {
        const users = await createTestUsers(3);
        const userIds = users.map(u => u.id);

        const response = await app.inject({
          method: 'POST',
          url: '/api/admin/users/batch-recharge',
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: { userIds, credits: 50 },
        });

        expect(response.statusCode).toBe(200);

        for (const id of userIds) {
          const user = await UserModel.findById(id);
          expect(user.credits).toBe(50);
        }
      });
    });
  });
});
```

## 总结

### 关键要点

1. **系统化方法**: 先梳理字段 → 设计测试 → 编写代码 → 生成报告
2. **完整覆盖**: 字段、API、业务逻辑、边界条件
3. **真实测试**: 使用真实 HTTP、真实数据库、真实中间件
4. **数据隔离**: 每个测试独立，前后清理
5. **统计分析**: 量化测试覆盖率，发现潜在问题
