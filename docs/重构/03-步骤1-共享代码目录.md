## 1. 目标

将管理端和机器端共享的代码提取到 `src/shared/` 目录，为后续分离做准备。

### 1.1 需要移动的内容

| 原路径 | 新路径 | 说明 |
|--------|--------|------|
| `src/protos/` | `src/shared/protos/` | gRPC proto 文件 |
| `src/types/` | `src/shared/types/` | 共享类型定义 |
| `src/utils/logger.ts` | `src/shared/utils/logger.ts` | 日志工具 |

---

## 2. 详细步骤

### 步骤 1.1: 创建目录结构

```bash
# 创建共享代码目录
mkdir -p src/shared/protos
mkdir -p src/shared/types
mkdir -p src/shared/utils
```

**预期结果:**
```
src/shared/
├── protos/          # gRPC proto 文件
├── types/           # 共享类型定义
└── utils/           # 共享工具函数
```

### 步骤 1.2: 移动 proto 文件

```bash
# 移动 proto 文件
```


### 步骤 1.5: 更新 tsconfig.json 路径映射

首先更新 TypeScript 配置，添加 `@shared` 路径别名：

**修改文件: `tsconfig.json`**

```json
{
  "compilerOptions": {
    "baseUrl": ".",
    "paths": {
      "@/*": ["src/*"],
      "@shared/*": ["src/shared/*"],
      "@manager/*": ["src/manager/*"],
      "@machine/*": ["src/machine/*"]
    }
  }
}
```

### 步骤 1.7: 批量替换 import 路径

#### 方式一: 使用 @别名 (推荐)

```bash
# 替换为 @shared 别名 (推荐)
find src -name "*.ts" -type f -exec sed -i '' "s|from '../utils/logger|from '@shared/utils/logger|g" {} \;
find src -name "*.ts" -type f -exec sed -i '' "s|from './utils/logger|from '@shared/utils/logger|g" {} \;
find src -name "*.ts" -type f -exec sed -i '' "s|from \"../utils/logger|from \"@shared/utils/logger|g" {} \;
find src -name "*.ts" -type f -exec sed -i '' "s|from \"./utils/logger|from \"@shared/utils/logger|g" {} \;

find src -name "*.ts" -type f -exec sed -i '' "s|from '../protos/|from '@shared/protos/|g" {} \;
find src -name "*.ts" -type f -exec sed -i '' "s|from './protos/|from '@shared/protos/|g" {} \;
find src -name "*.ts" -type f -exec sed -i '' "s|from \"../protos/|from \"@shared/protos/|g" {} \;
find src -name "*.ts" -type f -exec sed -i '' "s|from \"./protos/|from \"@shared/protos/|g" {} \;

find src -name "*.ts" -type f -exec sed -i '' "s|from '../types/|from '@shared/types/|g" {} \;
find src -name "*.ts" -type f -exec sed -i '' "s|from './types/|from '@shared/types/|g" {} \;
find src -name "*.ts" -type f -exec sed -i '' "s|from \"../types/|from \"@shared/types/|g" {} \;
find src -name "*.ts" -type f -exec sed -i '' "s|from \"./types/|from \"@shared/types/|g" {} \;
```

## 3. 验证测试

### 3.1 创建验证脚本

**新建文件: `tests/verify-step1-shared.js`**

```javascript
/**
 * 步骤1验证: 共享代码目录结构
 */
import { describe, it } from 'node:test';
import { strict as assert } from 'node:assert';
import fs from 'fs';
import path from 'path';

describe('Step 1: Shared Code Structure', () => {
  const sharedDir = path.join(process.cwd(), 'src/shared');

  it('should have shared directory', () => {
    assert.equal(fs.existsSync(sharedDir), true, 'src/shared directory should exist');
  });

  it('should have protos subdirectory', () => {
    const protoDir = path.join(sharedDir, 'protos');
    assert.equal(fs.existsSync(protoDir), true, 'src/shared/protos should exist');
  });

  it('should have types subdirectory', () => {
    const typesDir = path.join(sharedDir, 'types');
    assert.equal(fs.existsSync(typesDir), true, 'src/shared/types should exist');
  });

  it('should have utils subdirectory', () => {
    const utilsDir = path.join(sharedDir, 'utils');
    assert.equal(fs.existsSync(utilsDir), true, 'src/shared/utils should exist');
  });

  it('should have proto files in shared/protos', () => {
    const protoDir = path.join(sharedDir, 'protos');
    const files = fs.readdirSync(protoDir);
    assert.ok(files.length > 0, 'Should have proto files');
    assert.ok(files.some(f => f.endsWith('.proto')), 'Should have .proto file');
  });

  it('should have logger in shared/utils', () => {
    const loggerPath = path.join(sharedDir, 'utils', 'logger.ts');
    assert.equal(fs.existsSync(loggerPath), true, 'logger.ts should exist in shared/utils');
  });

  it('should have type files in shared/types', () => {
    const typesDir = path.join(sharedDir, 'types');
    const files = fs.readdirSync(typesDir);
    assert.ok(files.length > 0, 'Should have type files');
  });

  it('should not have old protos directory', () => {
    const oldProtoDir = path.join(process.cwd(), 'src/protos');
    assert.equal(fs.existsSync(oldProtoDir), false, 'src/protos should not exist');
  });

  it('should not have old types directory', () => {
    const oldTypesDir = path.join(process.cwd(), 'src/types');
    assert.equal(fs.existsSync(oldTypesDir), false, 'src/types should not exist');
  });

  it('should have correct import paths in manager files', () => {
    const appPath = path.join(process.cwd(), 'src/app.ts');
    const content = fs.readFileSync(appPath, 'utf-8');
    assert.ok(content.includes('shared/utils/logger'), 'app.ts should import from shared/utils/logger');
  });

  it('should have correct import paths in machine files', () => {
    const grpcPath = path.join(process.cwd(), 'src/machine/grpc.service.ts');
    const content = fs.readFileSync(grpcPath, 'utf-8');
    assert.ok(content.includes('shared/utils/logger'), 'grpc.service.ts should import from shared/utils/logger');
    assert.ok(content.includes('shared/protos'), 'grpc.service.ts should import from shared/protos');
  });
});
```
