### 步骤 3.1: 创建机器端 app.ts

**新建文件: `src/machine/app.ts`**

从 `src/machine/index.ts` 提取 MachineServer 类的逻辑：

```typescript
import { CONFIG } from './config.js';
import { logger } from '../utils/logger.js';
import { browserService } from './browser.service.js';
import { proxyService } from './proxy.service.js';
import grpcService, { grpcClient, startGrpcServer } from './grpc.service.js';

// 机器端状态枚举
export enum MachineState {
  STARTING = 'starting',
  RUNNING = 'running',
  RECONNECTING = 'reconnecting',
  SHUTTING_DOWN = 'shutting_down',
  STOPPED = 'stopped',
}

/**
 * 机器端主入口类
 * 负责启动所有服务并处理进程退出
 */
class MachineServer {
  // 当前机器端状态
  private state: MachineState = MachineState.STOPPED;

  // 重连定时器
  private reconnectTimer: NodeJS.Timeout | null = null;

  // 冷却定时器
  private cooldownTimer: NodeJS.Timeout | null = null;

  // 是否处于冷却期
  private inCooldown: boolean = false;

  // 冷却期时间（毫秒）
  private readonly COOLDOWN_PERIOD: number = 60000; // 1分钟

  /**
   * 获取当前状态
   */
  getState(): MachineState {
    return this.state;
  }

  /**
   * 设置状态
   */
  private setState(newState: MachineState): void {
    const oldState = this.state;
    this.state = newState;
    logger.info(`机器端状态变更: ${oldState} -> ${newState}`);
  }

  /**
   * 启动机器端
   */
  async start() {
    try {
      this.setState(MachineState.STARTING);

      logger.info('机器端配置:', {
        machineId: CONFIG.machineId,
        machineName: CONFIG.machineName,
        managerHost: CONFIG.managerHost,
        proxyPort: CONFIG.proxyPort,
        grpcPort: CONFIG.grpcPort,
      });

      // 启动 gRPC 服务器
      startGrpcServer(CONFIG.grpcPort);

      // 启动代理服务器
      proxyService.start();

      // 注册机器
      await grpcClient.register();

      // 连接到管理端
      await grpcClient.connect();

      // 处理进程退出
      this.setupProcessHandlers();

      this.setState(MachineState.RUNNING);
      logger.info('机器端启动完成');
    } catch (error) {
      logger.error('启动机器端失败:', error);
      this.setState(MachineState.STOPPED);
      process.exit(1);
    }
  }

  /**
   * 停止机器端
   */
  async stop() {
    try {
      if (this.state === MachineState.SHUTTING_DOWN || this.state === MachineState.STOPPED) {
        logger.info(`机器端已经在${this.state === MachineState.SHUTTING_DOWN ? '停止中' : '停止状态'}，不需要重复停止`);
        return;
      }

      this.setState(MachineState.SHUTTING_DOWN);
      logger.info('正在停止机器端...');

      if (this.reconnectTimer) {
        clearTimeout(this.reconnectTimer);
        this.reconnectTimer = null;
      }

      if (this.cooldownTimer) {
        clearTimeout(this.cooldownTimer);
        this.cooldownTimer = null;
      }

      await browserService.closeAllBrowsers();
      await proxyService.stop();

      this.setState(MachineState.STOPPED);
      logger.info('机器端已停止');
    } catch (error) {
      logger.error('停止机器端失败:', error);
      this.setState(MachineState.STOPPED);
      process.exit(1);
    }
  }

  /**
   * 设置进程处理程序
   */
  private setupProcessHandlers() {
    process.on('SIGINT', this.handleExit.bind(this));
    process.on('SIGTERM', this.handleExit.bind(this));
    process.on('uncaughtException', this.handleUncaughtException.bind(this));
    process.on('unhandledRejection', this.handleUnhandledRejection.bind(this));
  }

  /**
   * 处理进程退出
   */
  private async handleExit() {
    logger.info('收到退出信号，正在优雅地关闭...');
    await this.stop();
    process.exit(0);
  }

  /**
   * 处理未捕获的异常（含重连逻辑）
   */
  private handleUncaughtException(error: Error) {
    logger.error('未捕获的异常:', error);

    if (this.state === MachineState.SHUTTING_DOWN || this.state === MachineState.STOPPED) {
      logger.warn(`机器端当前状态为 ${this.state}，忽略连接错误`);
      return;
    }

    const isGrpcConnectionError = error.message && (
      error.message.includes('UNAVAILABLE: Connection dropped') ||
      error.message.includes('UNAVAILABLE: No connection established') ||
      error.message.includes('ECONNREFUSED') ||
      error.message.includes('UNAVAILABLE')
    );

    if (isGrpcConnectionError) {
      logger.warn('检测到 gRPC 连接错误，将尝试重新连接而不是停止服务');
      this.setState(MachineState.RECONNECTING);
      // 重连逻辑...
      return;
    }

    this.stop().then(() => process.exit(1));
  }

  /**
   * 处理未处理的拒绝
   */
  private handleUnhandledRejection(reason: any) {
    logger.error('未处理的拒绝:', reason);

    if (reason instanceof Error) {
      this.handleUncaughtException(reason);
      return;
    }

    if (this.state === MachineState.SHUTTING_DOWN || this.state === MachineState.STOPPED) {
      logger.warn(`机器端当前状态为 ${this.state}，忽略连接错误`);
      return;
    }

    const isGrpcConnectionError = reason && reason.message && (
      reason.message.includes('UNAVAILABLE: Connection dropped') ||
      reason.message.includes('UNAVAILABLE: No connection established') ||
      reason.message.includes('ECONNREFUSED') ||
      reason.message.includes('UNAVAILABLE')
    );

    if (isGrpcConnectionError) {
      const error = new Error(reason.message || 'gRPC connection error');
      this.handleUncaughtException(error);
    }
  }
}

/**
 * 启动机器端
 */
export async function startMachine() {
  const machineServer = new MachineServer();
  await machineServer.start();
}

/**
 * 构建机器端实例（用于测试）
 */
export async function buildMachine() {
  return new MachineServer();
}

export { MachineServer };
export default { MachineServer, startMachine, buildMachine };
```

### 3.1 创建验证脚本

```javascript
/**
 * 步骤3验证: 机器端入口重构
 */
import { describe, it, before, after } from 'node:test';
import { strict as assert } from 'node:assert';
import fs from 'fs';
import path from 'path';
import { spawn } from 'child_process';

describe('Step 3: Machine Entry Point Refactor', () => {
  const projectDir = process.cwd();
  const machineDir = path.join(projectDir, 'src/machine');
  let machineProcess = null;

  describe('1. File Structure', () => {
    it('should have machine app.ts', () => {
      const appPath = path.join(machineDir, 'app.ts');
      assert.equal(fs.existsSync(appPath), true, 'src/machine/app.ts should exist');
    });

    it('should have machine server.ts', () => {
      const serverPath = path.join(machineDir, 'server.ts');
      assert.equal(fs.existsSync(serverPath), true, 'src/machine/server.ts should exist');
    });

    it('should have backward compatible index.ts', () => {
      const indexPath = path.join(machineDir, 'index.ts');
      assert.equal(fs.existsSync(indexPath), true, 'src/machine/index.ts should exist');
    });
  });

  describe('2. Export Functions', () => {
    it('should export startMachine function', async () => {
      const appModule = await import(path.join(projectDir, 'src/machine/app.js'));
      assert.equal(typeof appModule.startMachine, 'function', 'should export startMachine');
    });

    it('should export buildMachine function', async () => {
      const appModule = await import(path.join(projectDir, 'src/machine/app.js'));
      assert.equal(typeof appModule.buildMachine, 'function', 'should export buildMachine');
    });

    it('should export MachineServer class', async () => {
      const appModule = await import(path.join(projectDir, 'src/machine/app.js'));
      assert.ok(appModule.MachineServer, 'should export MachineServer class');
    });
  });

  describe('3. Code Function', () => {
    it('should contain MachineState enum', async () => {
      const appModule = await import(path.join(projectDir, 'src/machine/app.js'));
      assert.ok(appModule.MachineState, 'should export MachineState enum');
    });

    it('MachineState should have correct values', async () => {
      const appModule = await import(path.join(projectDir, 'src/machine/app.js'));
      const { MachineState } = appModule;
      assert.equal(MachineState.STARTING, 'starting');
      assert.equal(MachineState.RUNNING, 'running');
      assert.equal(MachineState.STOPPED, 'stopped');
    });

    it('server.ts should import startMachine', () => {
      const serverPath = path.join(machineDir, 'server.ts');
      const content = fs.readFileSync(serverPath, 'utf-8');
      assert.ok(content.includes('startMachine'), 'server.ts should import startMachine');
    });

    it('index.ts should import from app.ts for backward compatibility', () => {
      const indexPath = path.join(machineDir, 'index.ts');
      const content = fs.readFileSync(indexPath, 'utf-8');
      assert.ok(content.includes('./app'), 'index.ts should import from ./app');
    });
  });

  describe('4. Startup Verification', () => {
    it('should start machine server', (t, done) => {
      machineProcess = spawn('pnpm', ['dev:machine'], {
        cwd: projectDir,
        stdio: ['ignore', 'pipe', 'pipe'],
        env: { ...process.env, NODE_ENV: 'test' }
      });

      let output = '';
      const timer = setTimeout(() => {
        machineProcess.kill();
        if (output.includes('gRPC') || output.includes('启动')) {
          assert.ok(true, 'Machine server started (partial success)');
          done();
        } else {
          done(new Error('Machine server start timeout'));
        }
      }, 12000);

      machineProcess.stdout.on('data', (data) => {
        output += data.toString();
        if (output.includes('机器端启动完成') || output.includes('gRPC 服务器已启动')) {
          clearTimeout(timer);
          assert.ok(true, 'Machine server started successfully');
          machineProcess.kill();
          done();
        }
      });

      machineProcess.stderr.on('data', (data) => {
        const error = data.toString();
        if (error.includes('ERROR')) {
          clearTimeout(timer);
          machineProcess.kill();
          done(new Error(`Machine server failed: ${error}`));
        }
      });
    }).timeout(15000);
  });

  after(() => {
    if (machineProcess && !machineProcess.killed) {
      machineProcess.kill();
    }
  });
});
```
